# v0.7.0 Safe Explorable Coast Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make TinyWorld’s coastline physically trustworthy: traversable coastal terrain, real swimmable water, safe recovery from falls, and a deterministic boundary between near-shore swimming and Tiny Boat-only outer-sea progression.

**Architecture:** Extend the existing `TraversalRules` domain rather than creating a second travel model. `BoundaryBuilder` creates actual `Workspace.Terrain` water plus a collidable seabed and returns explicit traversal bounds/safe recovery CFrames; a focused `TraversalSafetyService` observes server-side character positions and enforces safe recovery/boat-only outer sea without giving the client authority.

**Tech Stack:** Roblox/Luau, Rojo, Roblox Terrain API, existing pure-Luau test harness, shell source-contract guards.

## Global Constraints

- Preserve existing Tiny Boat ownership/profile semantics and the active profile compatibility bridge.
- Economy, progression, ownership and final traversal eligibility remain server-authoritative.
- Deterministic rules live in `src/shared`; Roblox world/service adapters live in `src/server`.
- Use test-first development for deterministic behaviour.
- Do not claim Studio/device acceptance from source inspection or CI.
- Do not add a framework or new runtime dependency.
- Do not implement Mermaid quests, pets, Tiny Car, full home progression or spaceship travel in this plan.

TraversalRules.Zone = {
	VillageLand = "village_land",
	ShoreSwim = "shore_swim",
	OuterSeaBoat = "outer_sea_boat",
}

function TraversalRules.classifyDistanceFromVillage(
	distance: number,
	shorelineDistance: number,
	swimDistance: number
): string
	if distance < shorelineDistance then
		return TraversalRules.Zone.VillageLand
	end
	if distance <= shorelineDistance + swimDistance then
		return TraversalRules.Zone.ShoreSwim
	end
	return TraversalRules.Zone.OuterSeaBoat
end

function TraversalRules.canOccupyZone(profile: any, zone: string): (boolean, string)
	if zone ~= TraversalRules.Zone.OuterSeaBoat then
		return true, "allowed"
	end
	if profile.ownsTinyBoat and profile.boatActive then
		return true, "allowed"
	end
	return false, "tiny_boat_required"
end
```

local Workspace = game:GetService("Workspace")
local Terrain = Workspace.Terrain

local SEA_DEPTH = 14
local SWIM_DISTANCE = 36

local function makeSea(parent: Instance, name: string, size: Vector3, position: Vector3)
	local waterSize = Vector3.new(size.X, SEA_DEPTH, size.Z)
	local waterCenter = Vector3.new(position.X, -(SEA_DEPTH / 2) + 0.5, position.Z)
	Terrain:FillBlock(CFrame.new(waterCenter), waterSize, Enum.Material.Water)

	local foam = makeDecoration(
		parent,
		name .. "Foam",
		Vector3.new(size.X * 0.9, 0.08, size.Z * 0.9),
		Vector3.new(position.X, 0.54, position.Z),
		VisualTheme.Colors.paper,
		Enum.Material.Neon
	)
	foam.Transparency = 0.9
end
```

- [ ] **Step 3: Implement focused server observer**

Create `src/server/TraversalSafetyService.luau` with this structure:

```luau
local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local shared = ReplicatedStorage:WaitForChild("TinyWorld"):WaitForChild("Shared")
local TraversalRules = require(shared:WaitForChild("TraversalRules"))

local TraversalSafetyService = {}
TraversalSafetyService.__index = TraversalSafetyService

local CHECK_INTERVAL = 0.2
local RECOVERY_Y = -20

function TraversalSafetyService.new(world, ProfileStore, PlayerStateService)
	local self = setmetatable({}, TraversalSafetyService)
	self.world = world
	self.ProfileStore = ProfileStore
	self.PlayerStateService = PlayerStateService
	self.tracked = {}
	self.accumulator = 0
	self.connection = RunService.Heartbeat:Connect(function(dt)
		self.accumulator += dt
		if self.accumulator < CHECK_INTERVAL then
			return
		end
		self.accumulator = 0
		self:_checkPlayers()
	end)
	return self
end

function TraversalSafetyService:trackPlayer(player: Player)
	self.tracked[player] = true
end

function TraversalSafetyService:removePlayer(player: Player)
	self.tracked[player] = nil
end

function TraversalSafetyService:_recover(player: Player, target: CFrame, message: string?)
	local character = player.Character
	if character then
		character:PivotTo(target)
	end
	if message then
		self.PlayerStateService.message(player, message)
	end
end

function TraversalSafetyService:_checkPlayers()
	for player in self.tracked do
		local character = player.Character
		local root = if character then character:FindFirstChild("HumanoidRootPart") else nil
		local profile = self.ProfileStore.get(player)
		if root and root:IsA("BasePart") and profile then
			if root.Position.Y < RECOVERY_Y then
				self:_recover(player, self.world.boundary.worldRecoveryCFrame)
				continue
			end

			local distance = math.max(math.abs(root.Position.X), math.abs(root.Position.Z))
			local zone = TraversalRules.classifyDistanceFromVillage(
				distance,
				self.world.boundary.shorelineDistance,
				self.world.boundary.swimDistance
			)
			local allowed, reason = TraversalRules.canOccupyZone(profile, zone)
			if not allowed and reason == "tiny_boat_required" then
				self:_recover(player, self.world.boundary.safeShoreCFrame, "The outer sea needs your Tiny Boat.")
			end
		end
	end
end

function TraversalSafetyService:stop()
	if self.connection then
		self.connection:Disconnect()
		self.connection = nil
	end
	table.clear(self.tracked)
end

return TraversalSafetyService
```
